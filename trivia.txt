byte b = (byte) 0xFF;
System.out.printf("%02X%n", b & 0xFF); // FF

System.out.println(Integer.toBinaryString(255));
// 11111111

int number = 0b11111111;
System.out.println(Integer.toBinaryString(number));

---

int number = 255;

System.out.printf("%02X%n", number); // FF
System.out.printf("%02x%n", number); // ff
System.out.printf("%02d%n", number); // 255

---

int number = 5;

System.out.println(
    String.format("%8s", Integer.toBinaryString(number))
          .replace(' ', '0')
);
